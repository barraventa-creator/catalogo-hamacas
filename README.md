let busqueda = '', categoriaActiva = 'Todos', sortDir = null, viewMode = 'grid';
const categorias = ['Todos', 'Ofertas', 'Hamacas Andinas', 'Sujetadores', 'Soportes', 'Instalación'];

function esc(s){
  return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function showToast() {
  DOM.toast.classList.add('show');
  setTimeout(() => DOM.toast.classList.remove('show'), 2600);
}

function buildCatBar() {
  DOM.catBar.innerHTML = categorias.map(c => `
    <button class="cat-tile${c === 'Todos' ? ' tile-todos' : ''}${c === categoriaActiva ? ' active' : ''}"
      data-cat="${esc(c)}">
      <span class="cat-tile-icon">${catIcons[c] || '📦'}</span>
      ${esc(c)}
    </button>`).join('');
}

function updateCatBar() {
  const tiles = DOM.catBar.querySelectorAll('.cat-tile');
  for (const t of tiles) t.classList.toggle('active', t.dataset.cat === categoriaActiva);
}

DOM.catBar.addEventListener('click', e => {
  const tile = e.target.closest('.cat-tile');
  if (!tile) return;
  categoriaActiva = tile.dataset.cat;
  updateCatBar();
  render(getFiltered());
});

function applySortUI() {
  DOM.sortBtn.classList.toggle('active', sortDir !== null);
  if (sortDir === 'asc') DOM.sortLabel.textContent = 'Menor precio';
  else if (sortDir === 'desc') DOM.sortLabel.textContent = 'Mayor precio';
  else DOM.sortLabel.textContent = 'Precio';
}

DOM.sortBtn.addEventListener('click', () => {
  sortDir = !sortDir ? 'asc' : sortDir === 'asc' ? 'desc' : null;
  applySortUI();
  render(getFiltered());
});

DOM.btnGrid.addEventListener('click', () => setView('grid'));
DOM.btnList.addEventListener('click', () => setView('list'));
DOM.btnLimpiar.addEventListener('click', () => {
  DOM.searchInput.value = '';
  busqueda = '';
  categoriaActiva = 'Todos';
  sortDir = null;
  updateCatBar();
  applySortUI();
  render(getFiltered());
});

function setView(m) {
  viewMode = m;
  DOM.btnGrid.classList.toggle('active', m === 'grid');
  DOM.btnList.classList.toggle('active', m === 'list');
  render(getFiltered());
}

function getFiltered() {
  const q = busqueda.toLowerCase().trim();
  let list = productos.filter(p => {
    if (categoriaActiva === 'Ofertas') {
      if (!p.priceOld && p.cat !== 'Ofertas') return false;
    } else if (categoriaActiva !== 'Todos' && p.cat !== categoriaActiva) {
      return false;
    }
    if (q && !p.code.toLowerCase().includes(q) && !p.name.toLowerCase().includes(q) && !p.cat.toLowerCase().includes(q)) return false;
    return true;
  });
  if (sortDir === 'asc') list.sort((a,b) => a.price - b.price);
  if (sortDir === 'desc') list.sort((a,b) => b.price - a.price);
  return list;
}

function render(list) {
  DOM.countNum.textContent = list.length;
  if (!list.length) {
    DOM.grid.innerHTML = '';
    DOM.empty.style.display = 'block';
    return;
  }
  DOM.empty.style.display = 'none';

  if (viewMode === 'list') {
    DOM.grid.className = 'grid list-mode';
    DOM.grid.innerHTML = list.map(p => {
      const pi = productos.indexOf(p);
      return `
        <div class="card list-card" data-idx="${pi}">
          <div class="card-img"><span>${p.emoji}</span></div>
          <div class="card-body">
            <div>
              <div class="card-code">${esc(p.code)} · ${esc(p.cat)}</div>
              <div class="card-name">${esc(p.name)}</div>
              <div class="card-medidas">${esc(p.medidas)}</div>
            </div>
            <div class="list-info">
              <div class="card-price"><small>S/ </small>${p.price}</div>
            </div>
          </div>
        </div>`;
    }).join('');
  } else {
    DOM.grid.className = 'grid';
    DOM.grid.innerHTML = list.map(p => {
      const pi = productos.indexOf(p);
      return `
        <div class="card" data-idx="${pi}">
          <div class="card-img">
            ${p.priceOld ? `<div class="liq-badge-img">Oferta -S/ ${p.priceOld - p.price}</div>` : ''}
            <span class="card-emoji">${p.emoji}</span>
            <img src="${esc(p.img)}" alt="${esc(p.name)}" loading="lazy" onload="this.classList.add('loaded')" onerror="this.classList.add('error')">
            <div class="card-hover-overlay">
              <div class="card-hover-label">${WA_SVG} Ver detalle</div>
            </div>
          </div>
          <div class="card-body">
            <div class="card-code">${esc(p.code)} · ${esc(p.cat)}</div>
            <div class="card-name">${esc(p.name)}</div>
            <div class="card-footer">
              <div class="card-price">
                ${p.priceOld ? `<span class="price-old">S/ <s>${p.priceOld}</s></span>` : ''}
                <span class="${p.priceOld ? 'price-discount' : ''}"><small>S/ </small>${p.price}</span>
              </div>
              <div class="card-medidas">${esc(p.resistencia)}</div>
            </div>
          </div>
        </div>`;
    }).join('');
  }
}

function openModal(idx) {
  const p = productos[idx];
  DOM.mName.textContent = p.name;
  DOM.mPrice.innerHTML = p.priceOld
    ? `<span style="font-size:1rem;color:var(--muted);text-decoration:line-through">S/ ${p.priceOld}</span> <span style="color:#b91c1c"><small>S/ </small>${p.price}</span>`
    : `<small style="font-size:1rem;font-weight:300;color:var(--muted)">S/ </small>${p.price}`;

  DOM.mSpecs.innerHTML = `
    <div class="spec"><div class="spec-l">Código</div><div class="spec-v">${esc(p.code)}</div></div>
    <div class="spec"><div class="spec-l">Categoría</div><div class="spec-v">${esc(p.cat)}</div></div>
    <div class="spec"><div class="spec-l">Medidas / Capacidad</div><div class="spec-v">${esc(p.medidas)}</div></div>
    <div class="spec"><div class="spec-l">Resistencia de peso</div><div class="spec-v">${esc(p.resistencia)}</div></div>
    <div class="spec" style="grid-column:span 2"><div class="spec-l">Material / Fabricación</div><div class="spec-v">${esc(p.material)}</div></div>
    ${p.colores && p.colores.length ? `<div class="spec" style="grid-column:span 2"><div class="spec-l">Colores / Variantes</div><div class="spec-v">${esc(p.colores.join(', '))}</div></div>` : ''}
  `;

  DOM.mImg.classList.remove('loaded');
  DOM.mImg.src = p.img;
  DOM.mImg.onload = () => DOM.mImg.classList.add('loaded');
  DOM.mImg.onerror = () => { DOM.mImg.style.display = 'none'; };

  DOM.mWA.onclick = () => {
    showToast();
    const msg = encodeURIComponent(
      `Hola Hamacas Lizaraso! 👋 Vi en su catálogo web (hamacasperu.com) el producto:\n\n` +
      `📦 *${p.name}* (Código: ${p.code})\n` +
      `💰 Precio: S/ ${p.price}\n\n` +
      `¿Tienen disponibilidad para coordinar envío o recojo en Miraflores? Gracias!`
    );
    window.open(`https://wa.me/${WA_NUMBER}?text=${msg}`, '_blank');
  };

  DOM.overlay.classList.add('open');
}
